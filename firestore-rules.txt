rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isOwnerEmail() {
      return signedIn()
        && request.auth.token.email == 'uu8832sr@gmail.com';
    }

    function isAdmin() {
      return signedIn() && (
        isOwnerEmail()
        || request.auth.uid == 'rN911472GUXjc1IYToSPhgf6zbs2'
        || request.auth.uid == 'iRdFBpDpXcNklIQ2tzP8kP1lDu42'
        || request.auth.uid == 'azMaRtblP0VevIvEQQDG6OZj5wh1'
        || (
          exists(/databases/$(database)/documents/admins/$(request.auth.uid))
          && get(/databases/$(database)/documents/admins/$(request.auth.uid)).data.enabled == true
        )
      );
    }

    function validPublicOrder(orderId) {
      return request.resource.data.orderId == orderId
        && request.resource.data.orderNo == orderId
        && request.resource.data.source is string
        && (request.resource.data.source == 'official-store' || request.resource.data.source == 'official-store-plate')
        && request.resource.data.customerName is string
        && request.resource.data.customerName.size() >= 1
        && request.resource.data.customerName.size() <= 40
        && request.resource.data.phone is string
        && request.resource.data.phone.size() >= 9
        && request.resource.data.phone.size() <= 20
        && request.resource.data.address is string
        && request.resource.data.address.size() >= 3
        && request.resource.data.address.size() <= 180
        && request.resource.data.model is string
        && request.resource.data.price is int
        && request.resource.data.price >= 0
        && request.resource.data.price <= 300000
        && request.resource.data.createdAt == request.time;
    }

    function validInstallmentLead() {
      return request.resource.data.eventType is string
        && (request.resource.data.eventType == 'trial' || request.resource.data.eventType == 'line_intent')
        && request.resource.data.sessionId is string
        && request.resource.data.sessionId.size() >= 8
        && request.resource.data.sessionId.size() <= 80
        && request.resource.data.phone is string
        && request.resource.data.phone.matches('^09[0-9]{8}$')
        && request.resource.data.modelId is string
        && request.resource.data.modelName is string
        && request.resource.data.style is string
        && request.resource.data.batteryKey is string
        && (request.resource.data.batteryKey == 'lead' || request.resource.data.batteryKey == 'ternary' || request.resource.data.batteryKey == 'lifepo4')
        && request.resource.data.batteryLabel is string
        && request.resource.data.price is int
        && request.resource.data.price >= 0
        && request.resource.data.price <= 300000
        && request.resource.data.term is int
        && (request.resource.data.term == 3 || request.resource.data.term == 6 || request.resource.data.term == 9 || request.resource.data.term == 12 || request.resource.data.term == 18 || request.resource.data.term == 24 || request.resource.data.term == 30 || request.resource.data.term == 36)
        && request.resource.data.monthly is int
        && request.resource.data.monthly >= 0
        && request.resource.data.total is int
        && request.resource.data.total >= 0
        && request.resource.data.eligibilityChecked is bool
        && request.resource.data.source == 'website-installment-free-v327'
        && request.resource.data.page is string
        && request.resource.data.userAgent is string
        && request.resource.data.createdAt == request.time;
    }

    match /admins/{uid} {
      allow get: if signedIn() && request.auth.uid == uid;
      allow list, create, update, delete: if false;
    }

    match /products/{productId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /siteSettings/{docId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /deliveryCases/{caseId} {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    match /orders/{orderId} {
      allow read, update, delete: if isAdmin();
      allow create: if isAdmin() || validPublicOrder(orderId);
    }

    match /onlineOrders/{orderId} {
      allow read, update, delete: if isAdmin();
      allow create: if isAdmin() || validPublicOrder(orderId);
    }

    match /installmentLeads/{leadId} {
      allow create: if validInstallmentLead();
      allow read, update, delete: if isAdmin();
    }

    match /warranties/{warrantyId} {
      allow get: if isAdmin() || resource.data.published == true;
      allow list: if isAdmin();
      allow create, update, delete: if isAdmin();
    }

    match /receipts/{receiptId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /sales/{saleId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /campaigns/{campaignId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /drawCodes/{codeId} {
      allow read, create, update, delete: if isAdmin();
    }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
